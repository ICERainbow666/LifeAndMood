# 数据库设计 · 生活与心情

> 配套画面：login / register / DASHBOARD(HISTORY) / charts / diary / share / settings
> 数据库：MySQL 8.x（utf8mb4，支持 emoji 头像与心情图标存储）
> 说明：前端当前为 localStorage 演示态，本 SQL 用于接后端版本时直接可用。表结构按当前页面已实现的功能重新设计，并为宠物系统预留。

## 心情编码约定

| mood 值 | 图标 | 含义 | 前端 key（演示态） |
|---------|------|------|-------------------|
| 1 | 😊 | 开心 | happy |
| 2 | 😌 | 平静 | calm |
| 3 | 😔 | 低落 | low |
| 4 | 😤 | 烦躁 | annoyed |
| 5 | 😢 | 难过 | sad |

- 记账心情可为空（NULL = 未选择）；心情手札心情必填
- 分享帖子冗余存一份心情，避免查询多表关联

## 表清单

| 表名 | 说明 | 对应画面 |
|---|---|---|
| `user` | 用户账号（登录注册） | login / register |
| `category` | 收支分类（内置 + 用户自定义） | 记一笔 |
| `transaction` | 收支记录（核心表，带心情） | 记账日历 / 今日一览 / 收支与情绪 |
| `diary` | 心情手札（一天可多篇） | 心情手札 |
| `share_post` | 分享帖子（心情帖 / 账单帖） | 分享广场 |
| `share_like` | 点赞记录 | 分享广场 |
| `budget` | 月度预算 | 个人设置 / 今日一览 |
| `pet` | 宠物（预留） | 右下角宠物 |
| `pet_care_log` | 宠物喂养记录（预留） | 宠物系统 |
| `savings_goal` | 攒钱目标（预留） | 攒钱搭子 |
| `savings_member` | 小队成员（预留） | 攒钱搭子 |
| `savings_log` | 投币记录（预留） | 攒钱搭子 |
| `message` | 站内消息（预留） | 顶栏消息中心 |

## 建表语句

```sql
-- 创建数据库
CREATE DATABASE IF NOT EXISTS lifeandmood
    DEFAULT CHARACTER SET utf8mb4
    DEFAULT COLLATE utf8mb4_unicode_ci;

USE lifeandmood;

-- 1. 用户表（login / register，所有业务数据按 user_id 隔离）
CREATE TABLE user (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    username    VARCHAR(16)  NOT NULL COMMENT '登录名，2-16 字符，注册页校验',
    password    VARCHAR(64)  NOT NULL COMMENT '密码（后端加密存储，演示态为明文）',
    nickname    VARCHAR(16)  DEFAULT '' COMMENT '昵称，默认取用户名',
    avatar      VARCHAR(8)   DEFAULT '👤' COMMENT '头像 emoji，注册时随机分配',
    status      TINYINT      DEFAULT 1 COMMENT '1=正常 0=禁用',
    created_at  DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '注册时间',
    updated_at  DATETIME     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    UNIQUE KEY uk_user_username (username)
) COMMENT '用户表';

-- 2. 分类表（支出 8 类 + 收入 6 类内置，与记一笔弹窗固定选项一致）
CREATE TABLE category (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    user_id     INT          NOT NULL DEFAULT 0 COMMENT '所属用户，0=内置分类（全员可用）',
    name        VARCHAR(20)  NOT NULL COMMENT '分类名称，如：餐饮、交通',
    type        TINYINT      NOT NULL COMMENT '1=支出 2=收入',
    icon        VARCHAR(10)  DEFAULT '' COMMENT '图标 emoji，如 🍜',
    is_default  TINYINT(1)   DEFAULT 0 COMMENT '1=内置分类 0=用户自定义',
    created_at  DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    UNIQUE KEY uk_cat_user_type_name (user_id, type, name)
    -- ,CONSTRAINT fk_cat_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE 取消外键 我要userid = 0 来获取默认值
) COMMENT '收支分类表';
-- 唯一键含 type：同一用户可以建同名但不同类型的分类（如收入"红包"和支出"红包"）；
-- user_id 用 0 而非 NULL，避免 NULL 不参与唯一索引查重导致内置分类重名

-- 3. 收支记录表（核心表，每笔消费可带心情；对应记账日历的记一笔）
CREATE TABLE transaction (
    id           BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id      INT            NOT NULL COMMENT '所属用户',
    type         TINYINT        NOT NULL COMMENT '1=支出 2=收入',
    category_id  INT            NOT NULL COMMENT '分类id（支出/收入分类须与 type 匹配）',
    amount       DECIMAL(12,2)  NOT NULL COMMENT '金额，>0，记账弹窗校验',
    record_date  DATE           NOT NULL COMMENT '记账日期（日历格子定位）',
    mood         TINYINT        DEFAULT NULL COMMENT '记账时心情：1😊 2😌 3😔 4😤 5😢，可为空',
    note         VARCHAR(200)   DEFAULT '' COMMENT '备注（记账弹窗"备注"栏）',
    created_at   DATETIME       DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    CONSTRAINT fk_txn_user     FOREIGN KEY (user_id)     REFERENCES user(id)     ON DELETE CASCADE,
    CONSTRAINT fk_txn_category FOREIGN KEY (category_id) REFERENCES category(id),
    INDEX idx_txn_user_date (user_id, record_date),
    INDEX idx_txn_mood (mood)
) COMMENT '收支记录表';

-- 4. 心情手札表（一天可多篇，编辑走 updated_at；对应心情手札页）
CREATE TABLE diary (
    id          BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id     INT      NOT NULL COMMENT '所属用户',
    diary_date  DATE     NOT NULL COMMENT '日记日期',
    mood        TINYINT  NOT NULL COMMENT '当天心情：1😊 2😌 3😔 4😤 5😢（必填）',
    content     TEXT COMMENT '日记正文',
    created_at  DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at  DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间（列表按此倒序）',
    CONSTRAINT fk_diary_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE,
    INDEX idx_diary_user_date (user_id, diary_date)
) COMMENT '心情手札表';

-- 5. 分享帖子表（分享广场；内容为快照，源记录删除不影响已发布帖子）
CREATE TABLE share_post (
    id              BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id         INT           NOT NULL COMMENT '发布者',
    kind            TINYINT       NOT NULL COMMENT '1=心情帖（源：diary） 2=账单帖（源：transaction）',
    diary_id        BIGINT        DEFAULT NULL COMMENT '来源手札id（kind=1 时有值）',
    transaction_id  BIGINT        DEFAULT NULL COMMENT '来源账单id（kind=2 时有值）',
    mood            TINYINT       DEFAULT NULL COMMENT '发布时心情（冗余快照）',
    content         VARCHAR(500)  DEFAULT '' COMMENT '分享文字（快照；账单帖存备注或分类名）',
    amount          DECIMAL(12,2) DEFAULT NULL COMMENT '账单帖金额（规划：可配置脱敏隐藏）',
    is_anonymous    TINYINT(1)    DEFAULT 0 COMMENT '1=匿名发布（规划：只显示宠物头像）',
    like_count      INT           DEFAULT 0 COMMENT '点赞数（冗余计数，与 share_like 同步维护）',
    status          TINYINT       DEFAULT 3 COMMENT '0=已删除 1=正常（审核通过） 2=被举报屏蔽 3=待审核（发帖默认） 4=审核未通过',
    created_at      DATETIME      DEFAULT CURRENT_TIMESTAMP COMMENT '发布时间（广场按此倒序）',
    CONSTRAINT fk_post_user   FOREIGN KEY (user_id)        REFERENCES user(id)      ON DELETE CASCADE,
    CONSTRAINT fk_post_diary  FOREIGN KEY (diary_id)       REFERENCES diary(id)     ON DELETE SET NULL,
    CONSTRAINT fk_post_txn    FOREIGN KEY (transaction_id) REFERENCES transaction(id) ON DELETE SET NULL,
    INDEX idx_post_time (status, created_at)
) COMMENT '分享帖子表';

-- 6. 点赞记录表（一人一帖一赞，再点=取消）
CREATE TABLE share_like (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    post_id     BIGINT   NOT NULL COMMENT '帖子id',
    user_id     INT      NOT NULL COMMENT '点赞人',
    created_at  DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '点赞时间',
    UNIQUE KEY uk_like_once (post_id, user_id),
    CONSTRAINT fk_like_post FOREIGN KEY (post_id) REFERENCES share_post(id) ON DELETE CASCADE,
    CONSTRAINT fk_like_user FOREIGN KEY (user_id) REFERENCES user(id)     ON DELETE CASCADE
) COMMENT '分享点赞表';

-- 7. 月度预算表（settings 页设置，每月一条）
CREATE TABLE budget (
    id           INT AUTO_INCREMENT PRIMARY KEY,
    user_id      INT           NOT NULL COMMENT '所属用户',
    budget_month CHAR(7)       NOT NULL COMMENT '月份，格式 2026-09',
    amount       DECIMAL(12,2) NOT NULL COMMENT '本月预算总额',
    UNIQUE KEY uk_budget_user_month (user_id, budget_month),
    CONSTRAINT fk_budget_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE
) COMMENT '月度预算表';

-- 8. 宠物表（预留：右下角宠物组件的养成数据；记账 = 喂养）
CREATE TABLE pet (
    id            INT AUTO_INCREMENT PRIMARY KEY,
    user_id       INT          NOT NULL COMMENT '宠物主人，一人一只',
    name          VARCHAR(20)  DEFAULT '咕噜' COMMENT '宠物名',
    species       TINYINT      DEFAULT 1 COMMENT '种类（预留：1=猫 2=狗 …）',
    skin          VARCHAR(8)   DEFAULT '🐱' COMMENT '当前外观 emoji（随等级/心情变化）',
    level         INT          DEFAULT 1 COMMENT '等级（连续记账升级）',
    exp           INT          DEFAULT 0 COMMENT '当前经验值',
    satiety       INT          DEFAULT 100 COMMENT '饱食度 0-100，疏于记录会下降但不会死亡',
    is_alive      TINYINT(1)   DEFAULT 1 COMMENT '1=在世（产品设定：永不死亡）',
    updated_at    DATETIME     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    UNIQUE KEY uk_pet_user (user_id),
    CONSTRAINT fk_pet_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE
) COMMENT '宠物表（预留）';

-- 9. 宠物喂养记录表（预留：每记一笔账 = 喂一次，用于连续记账统计）
CREATE TABLE pet_care_log (
    id            INT AUTO_INCREMENT PRIMARY KEY,
    pet_id        INT      NOT NULL COMMENT '宠物id',
    feed_date     DATE     NOT NULL COMMENT '喂养日期（每天多次喂养只计首次为口粮）',
    source_txn_id BIGINT   DEFAULT NULL COMMENT '触发喂养的账单id',
    exp_gained    INT      DEFAULT 10 COMMENT '本次获得经验',
    created_at    DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '喂养时间',
    CONSTRAINT fk_care_pet FOREIGN KEY (pet_id)        REFERENCES pet(id)         ON DELETE CASCADE,
    CONSTRAINT fk_care_txn FOREIGN KEY (source_txn_id) REFERENCES transaction(id) ON DELETE SET NULL,
    UNIQUE KEY uk_feed_once (pet_id, feed_date)
) COMMENT '宠物喂养记录表（预留）';

-- 10. 攒钱目标表（预留：2-6 人小队共同目标，只记进度不做资金流转）
CREATE TABLE savings_goal (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    creator_id  INT          NOT NULL COMMENT '创建者',
    name        VARCHAR(30)  NOT NULL COMMENT '目标名，如：一起去大理旅行',
    target_amt  DECIMAL(12,2) NOT NULL COMMENT '目标金额',
    deadline    DATE         DEFAULT NULL COMMENT '截止日期（可选）',
    invite_code CHAR(6)      NOT NULL COMMENT '6位邀请码，全员可见',
    celebrated  TINYINT(1)   DEFAULT 0 COMMENT '1=已触发达成庆祝',
    created_at  DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    UNIQUE KEY uk_invite_code (invite_code),
    CONSTRAINT fk_goal_creator FOREIGN KEY (creator_id) REFERENCES user(id) ON DELETE CASCADE
) COMMENT '攒钱目标表（预留）';

-- 11. 小队成员表（预留：创建者+受邀搭子，2-6 人）
CREATE TABLE savings_member (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    goal_id     INT          NOT NULL COMMENT '所属目标',
    user_id     INT          NOT NULL COMMENT '成员用户',
    claimed_amt DECIMAL(12,2) DEFAULT 0 COMMENT '认领份额',
    joined_at   DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '加入时间',
    UNIQUE KEY uk_member_once (goal_id, user_id),
    CONSTRAINT fk_member_goal FOREIGN KEY (goal_id) REFERENCES savings_goal(id) ON DELETE CASCADE,
    CONSTRAINT fk_member_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE
) COMMENT '小队成员表（预留）';

-- 12. 投币记录表（预留：谁在哪天给哪个存钱罐投了多少）
CREATE TABLE savings_log (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    goal_id     INT           NOT NULL COMMENT '所属目标',
    user_id     INT           NOT NULL COMMENT '投币成员',
    amount      DECIMAL(12,2) NOT NULL COMMENT '本次投入金额',
    coin_date   DATE          NOT NULL COMMENT '投币日期',
    created_at  DATETIME      DEFAULT CURRENT_TIMESTAMP COMMENT '投币时间',
    CONSTRAINT fk_log_goal FOREIGN KEY (goal_id) REFERENCES savings_goal(id) ON DELETE CASCADE,
    CONSTRAINT fk_log_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE
) COMMENT '投币记录表（预留）';

-- 13. 站内消息表（预留：顶栏消息中心，接后端后由后端推送替代 localStorage）
CREATE TABLE message (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    user_id     INT          NOT NULL COMMENT '消息归属用户',
    type        VARCHAR(10)  NOT NULL COMMENT '类型：pet/remind/team/plaza/system',
    title       VARCHAR(100) NOT NULL COMMENT '消息内容',
    is_read     TINYINT(1)   DEFAULT 0 COMMENT '1=已读 0=未读',
    created_at  DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '推送时间',
    CONSTRAINT fk_msg_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE
) COMMENT '站内消息表（预留）';
```

## 常用查询示例（对应各画面）

```sql
-- 今日一览（DASHBOARD）：本月收支总览与结余
SELECT type, SUM(amount) AS total
FROM transaction
WHERE user_id = 1
  AND record_date BETWEEN '2026-09-01' AND '2026-09-30'
GROUP BY type;

-- 今日一览：预算使用进度（已花 / 预算）
SELECT b.amount AS budget,
       COALESCE(SUM(t.amount), 0) AS used
FROM budget b
LEFT JOIN transaction t
       ON t.user_id = b.user_id
      AND t.type = 1
      AND DATE_FORMAT(t.record_date, '%Y-%m') = b.budget_month
WHERE b.user_id = 1 AND b.budget_month = '2026-09'
GROUP BY b.amount;

-- 记账日历（HISTORY）：整月每天汇总（金额 + 笔数），渲染 42 格日历
SELECT record_date,
       SUM(IF(type = 2, amount, 0)) AS income,
       SUM(IF(type = 1, amount, 0)) AS expense,
       COUNT(*) AS cnt
FROM transaction
WHERE user_id = 1 AND record_date LIKE '2026-09%'
GROUP BY record_date;

-- 记账日历：某日明细（点击日历格子后的当日弹窗）
SELECT t.*, c.name AS category_name, c.icon
FROM transaction t JOIN category c ON t.category_id = c.id
WHERE t.user_id = 1 AND t.record_date = '2026-09-20'
ORDER BY t.created_at;

-- 记一笔（COMMAND_HISTORY_SAVE）
INSERT INTO transaction (user_id, type, category_id, amount, record_date, mood, note)
VALUES (1, 1, 1, 25.50, '2026-09-20', 4, '加班点外卖');

-- 心情手札（DIARY）：分页列表（按更新时间倒序，每页 5 条）
SELECT * FROM diary
WHERE user_id = 1
ORDER BY updated_at DESC
LIMIT 5 OFFSET 0;

-- 收支与情绪（CHARTS）：本月支出分类饼图
SELECT c.name, SUM(t.amount) AS total
FROM transaction t JOIN category c ON t.category_id = c.id
WHERE t.user_id = 1 AND t.type = 1 AND t.record_date LIKE '2026-09%'
GROUP BY c.name ORDER BY total DESC;

-- 收支与情绪：各心情的平均消费（心情 × 花销关联分析 + 情绪化消费预警）
SELECT mood, COUNT(*) AS cnt, AVG(amount) AS avg_spend, SUM(amount) AS total
FROM transaction
WHERE user_id = 1 AND type = 1 AND mood IS NOT NULL
GROUP BY mood;

-- 收支与情绪：6 个月收支趋势柱图
SELECT DATE_FORMAT(record_date, '%Y-%m') AS ym,
       SUM(IF(type = 2, amount, 0)) AS income,
       SUM(IF(type = 1, amount, 0)) AS expense
FROM transaction
WHERE user_id = 1 AND record_date >= DATE_SUB('2026-09-01', INTERVAL 5 MONTH)
GROUP BY ym ORDER BY ym;

-- 分享广场（PLAZA）：帖子流（正常状态，按发布时间倒序）+ 发布者信息
SELECT p.*, u.nickname, u.avatar
FROM share_post p JOIN user u ON p.user_id = u.id
WHERE p.status = 1
ORDER BY p.created_at DESC
LIMIT 20 OFFSET 0;

-- 分享广场：按类型筛选（1=心情帖 2=账单帖）
SELECT p.*, u.nickname, u.avatar
FROM share_post p JOIN user u ON p.user_id = u.id
WHERE p.status = 1 AND p.kind = 1
ORDER BY p.created_at DESC;

-- 分享广场：我是否已点赞（点亮 👍）
SELECT 1 FROM share_like
WHERE post_id = 1 AND user_id = 1;
```

## 表关系一览

```
user 1 ──── n transaction     （每笔账单属于某个用户）
user 1 ──── n diary           （一天可多篇手札）
user 1 ──── n share_post      （发帖）
user 1 ──── n share_like      （点赞）
user 1 ──── n budget          （每月一条预算）
user 1 ── 1 pet               （一人一只宠物，预留）
pet  1 ──── n pet_care_log    （每天首次记账=喂养，预留）
user 1 ──── n savings_goal    （创建的攒钱目标，预留）
savings_goal 1 ── n savings_member （小队 2-6 人，预留）
savings_goal 1 ── n savings_log     （一个罐子多笔投币，预留）
user 1 ──── n message         （一人多条消息，预留）

category 1 ── n transaction   （记录属于某个分类）
diary    1 ── n share_post    （心情帖来源，SET NULL：源删除帖子保留）
transaction 1 ── n share_post （账单帖来源，SET NULL：同上）
share_post 1 ── n share_like  （一帖多赞，一人一赞）
```
